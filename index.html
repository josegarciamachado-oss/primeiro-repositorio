class Jogo: def __init__(self, titulo, genero, preco, incluso_no_plano): self.titulo = titulo self.genero = genero self.preco = preco self.incluso_no_plano = incluso_no_plano # True se estiver na assinatura def exibir_detalhes(self): status_plano = "Incluso na Assinatura" if self.incluso_no_plano else "Apenas Compra Avulsa" return f"- {self.titulo} | Gênero: {self.genero} | Preço: R${self.preco:.2f} [{status_plano}]" class CatalogoJogos: def __init__(self, nome_empresa, preco_assinatura): self.nome_empresa = nome_empresa self.preco_assinatura = preco_assinatura self.jogos = [] def adicionar_jogo(self, jogo): self.jogos.append(jogo) def mostrar_servico(self): print(f"=== BEM-VINDO À {self.nome_empresa.upper()} ===") print(f"Como funciona nosso serviço:") print(f"1. Assinatura Mensal: Por R${self.preco_assinatura:.2f}/mês, você joga todos os títulos marcados como 'Incluso na Assinatura'.") print(f"2. Compra Avulsa: Você pode comprar qualquer jogo individualmente para ser seu para sempre.") print("-" * 50) def listar_catalogo(self): print("CATÁLOGO DE JOGOS DISPONÍVEIS:") for jogo in self.jogos: print(jogo.exibir_detalhes()) print("-" * 50) # --- Demonstração do uso --- # Criando a empresa de jogos empresa = CatalogoJogos(nome_empresa="PixelVerse Games", preco_assinatura=29.90) # Cadastrando jogos empresa.adicionar_jogo(Jogo("Cyber Odyssey", "RPG", 149.90, incluso_no_plano=True)) empresa.adicionar_jogo(Jogo("Speed Racers", "Corrida", 89.90, incluso_no_plano=True)) empresa.adicionar_jogo(Jogo("Galactic Legends", "Estratégia", 199.90, incluso_no_plano=False)) # Exibindo o serviço e o catálogo empresa.mostrar_servico() empresa.listar_catalogo()
